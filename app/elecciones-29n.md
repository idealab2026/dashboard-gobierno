const partyFilter=document.getElementById("il-29n-party-choice");
const topicFilter=document.getElementById("il-29n-topic-choice");
const electionTable=document.getElementById("il-29n-compare-table");
function updateElectionFilters(){
  const party=partyFilter.value;
  const topic=topicFilter.value;
  electionTable.querySelectorAll("[data-party]").forEach(el=>{
    el.classList.toggle("il-29n-col-hidden",party!=="all"&&el.dataset.party!==party);
  });
  electionTable.querySelectorAll("tbody tr[data-topic]").forEach(row=>{
    row.classList.toggle("il-29n-topic-hidden",topic!=="all"&&row.dataset.topic!==topic);
  });
}
partyFilter.addEventListener("change",updateElectionFilters);
topicFilter.addEventListener("change",updateElectionFilters);
updateElectionFilters();
